"use strict";
const {test}=require('node:test'),assert=require('node:assert/strict'),fs=require('fs'),os=require('os'),path=require('path');
const E=require('./engine'),{Store}=require('./store'),{Service}=require('./service');
const plan=(ids=['phoebe','carnet'],extra={})=>({accepted:true,reason:'Moyens suffisants dans la situation.',title:'Action précise',effect:'Un effet limité et vérifiable',requirements:ids.map(id=>({role:id,eligible:[id]})),delay:1,ongoing:false,risks:'Échec possible',...extra});
function key(s,ids=['phoebe','carnet'],target='root-phoebe'){return E.propose(s,'phoebe',{kind:'key',target,resource:ids[0],text:'Une action précise pour cet état'},plan(ids));}
function nextPlayer(s){s.day++;s.phase='player';s.spent={phoebe:false,commanditaire:false};}
function acquire(s,n){s.phase='morning';E.morning(s,{outcomes:[{id:n.id,success:true,reason:'Fait obtenu',losses:[],consume:[]}],states:[],facts:[]});}
test('un seul coup serveur, y compris plusieurs poses',()=>{const s=E.initial();key(s);assert.throws(()=>key(s,['nina','jules']),/déjà joué/);});
test('préparation progressive et résolution matinale',()=>{const s=E.initial(),n=key(s);assert.equal(n.status,'preparing');nextPlayer(s);E.place(s,'phoebe',n.id,'carnet');assert.equal(n.status,'ready');assert.equal(n.dueDay,3);assert.equal(E.resource(s,'phoebe').heldBy,n.id);acquire(s,n);assert.equal(n.status,'acquired');assert.equal(E.availability(s,E.resource(s,'phoebe')),'free');});
test('deux ressources minimum dans la voie principale',()=>{const s=E.initial();assert.throws(()=>key(s,['phoebe']),/au moins deux/);});
test('la pose peut réattribuer les rôles sans doublon',()=>{const s=E.initial();const p=plan(['phoebe','carnet']);p.requirements=[{role:'observer',eligible:['phoebe','carnet']},{role:'vision',eligible:['phoebe']}];const n=E.propose(s,'phoebe',{kind:'key',target:'root-phoebe',resource:'phoebe',text:'Une action expliquée précisément'},p);nextPlayer(s);E.place(s,'phoebe',n.id,'carnet');assert.equal(n.status,'ready');});
test('aucune duplication d’une ressource unique entre rôles',()=>{const s=E.initial();assert.throws(()=>E.propose(s,'phoebe',{kind:'key',target:'root-phoebe',resource:'phoebe',text:'Une action expliquée'},plan(['phoebe','phoebe'])),/même ressource/);});
test('plusieurs clés acquises restent retenues dans un sous-objectif',()=>{const s=E.initial();const sub=E.propose(s,'phoebe',{kind:'subgoal',target:'root-phoebe',text:'Trouver un lieu précis'},plan([], {title:'Lieu identifié'}));nextPlayer(s);const a=key(s,['phoebe'],sub.id);acquire(s,a);assert.equal(a.status,'acquired');assert.equal(E.availability(s,E.resource(s,'phoebe')),'engaged');const b=key(s,['nina'],sub.id);acquire(s,b);assert.equal(E.availability(s,E.resource(s,'nina')),'engaged');s.phase='morning';E.morning(s,{outcomes:[],states:[{id:sub.id,value:true,reason:'Les deux résultats suffisent'}],facts:[]});assert.equal(E.availability(s,E.resource(s,'phoebe')),'free');assert.equal(E.availability(s,E.resource(s,'nina')),'free');});
function lockedMain(){const s=E.initial(),k=key(s);nextPlayer(s);E.place(s,'phoebe',k.id,'carnet');acquire(s,k);s.phase='ai';const lock=E.propose(s,'commanditaire',{kind:'lock',target:k.id,resource:'gardien',text:'Le gardien empêche cet effet'},plan(['gardien','acces-galerie'],{ongoing:true}));s.day++;s.spent.commanditaire=false;E.place(s,'commanditaire',lock.id,'acces-galerie');acquire(s,lock);return {s,k,lock};}
test('verrou principal gèle les anciennes pièces libres',()=>{const {s,k}=lockedMain();assert.equal(E.availability(s,E.resource(s,'phoebe')),'frozen');assert.equal(k.status,'acquired');assert.throws(()=>key(s,['phoebe','nina']),/disponible/);});
test('engagement ailleurs protégé, gel différé à sa libération',()=>{const {s}=lockedMain(),r=E.resource(s,'phoebe');r.heldBy='autre';assert.equal(E.availability(s,r),'engaged');r.heldBy=null;assert.equal(E.availability(s,r),'frozen');});
test('on ne lève pas un verrou avec la pièce qu’il gèle',()=>{const {s,lock}=lockedMain();assert.throws(()=>key(s,['phoebe','nina'],lock.id),/disponible/);});
test('lever avec d’autres moyens libère les pièces gelées',()=>{const {s,lock}=lockedMain(),n=key(s,['nina','jules'],lock.id);nextPlayer(s);E.place(s,'phoebe',n.id,'jules');acquire(s,n);assert.equal(lock.status,'removed');assert.equal(E.availability(s,E.resource(s,'phoebe')),'free');});
test('verrou principal uniquement sur la dernière clé',()=>{const s=E.initial(),a=key(s);nextPlayer(s);const b=key(s,['nina','jules']);s.phase='ai';assert.throws(()=>E.targetCheck(s,'commanditaire','lock',a.id),/dernière/);assert.doesNotThrow(()=>E.targetCheck(s,'commanditaire','lock',b.id));assert.throws(()=>E.targetCheck(s,'commanditaire','lock','root-phoebe'),/dernière/);});

test('attaque complète résolue à date, objets consommés ne reviennent pas',()=>{const s=E.initial();const n=E.propose(s,'phoebe',{kind:'attack',target:'convocation',text:'Saisir la convocation adverse'},plan());nextPlayer(s);E.place(s,'phoebe',n.id,'phoebe');nextPlayer(s);E.place(s,'phoebe',n.id,'carnet');s.phase='morning';E.morning(s,{outcomes:[{id:n.id,success:true,reason:'Objet saisi',losses:['convocation'],consume:['carnet']}],states:[],facts:[]});assert.equal(n.status,'resolved');assert.equal(E.availability(s,E.resource(s,'phoebe')),'free');assert.equal(E.availability(s,E.resource(s,'carnet')),'lost');});
test('pas de déclaration tardive matériellement impossible',()=>{const s=E.initial();s.day=10;assert.throws(()=>E.propose(s,'phoebe',{kind:'attack',target:'gardien',text:'Neutraliser le gardien adverse'},plan()),/assez/);});
